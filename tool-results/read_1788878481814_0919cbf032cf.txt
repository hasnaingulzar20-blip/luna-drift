    47→function formatRemaining(sec: number) {
    48→  const m = Math.floor(sec / 60);
    49→  const s = sec % 60;
    50→  return `${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    51→}

    53→const SILENCE_LABEL = "Silence";
    54→
    55→function stepLabel(sound: SequenceSound): string {
    56→  return sound === "silence" ? SILENCE_LABEL : getSoundscape(sound).name;
    57→}

    59→/** A small room → minutes chip used in sequence flows. */
    60→function StepChip({
    61→  step,
    62→  state = "idle",
    63→  remaining,
    64→}: {
    65→  step: WindDownStep;
    66→  state?: "idle" | "done" | "current" | "ahead";
    67→  remaining?: number;
    68→}) {
    69→  const base =
    70→    "flex items-center gap-1.5 rounded-full px-3 py-1.5 text-[11px] transition-all duration-500 ring-1";
    71→  const look =
    72→    state === "current"
    73→      ? "bg-moon-200/15 text-moon-100 ring-moon-200/50 shadow-[0_0_18px_rgba(236,226,200,0.18)]"
    74→      : state === "done"
    75→        ? "bg-white/[0.02] text-mist-500 ring-white/6 line-through decoration-mist-600/60"
    76→        : state === "ahead"
    77→          ? "bg-white/[0.03] text-mist-300 ring-white/10"
    78→          : "bg-white/[0.03] text-mist-300 ring-white/10";
    79→  return (
    80→    <span className={`${base} ${look}`}>
    81→      {step.soundscape === "silence" ? (
    82→        <Moon className="h-3 w-3 text-mist-500" aria-hidden="true" />
    83→      ) : (
    84→        <span
    85→          aria-hidden="true"
    86→          className="h-1.5 w-1.5 rounded-full"
    87→          style={{ background: getSoundscape(step.soundscape as SoundscapeId).hue }}
    88→        />
    89→      )}
    90→      {stepLabel(step.soundscape)}
    91→      <span className="font-mono text-[10px] text-mist-500">{step.minutes}′</span>
    92→      {state === "current" && remaining !== undefined && (
    93→        <span className="font-mono text-[10px] text-moon-200" aria-live="polite">
    94→          {formatRemaining(remaining)}
    95→        </span>
    96→      )}
    97→    </span>
    98→  );
    99→}
