    27→const rnd = (min: number, max: number) => min + Math.random() * (max - min);
    28→const pick = <T,>(arr: T[]): T => arr[Math.floor(Math.random() * arr.length)];

    30→function makeNoiseBuffer(ctx: AudioContext, kind: "white" | "pink" | "brown", seconds = 4): AudioBuffer {
    31→  const length = ctx.sampleRate * seconds;
    32→  const buffer = ctx.createBuffer(1, length, ctx.sampleRate);
    33→  const data = buffer.getChannelData(0);
    34→  if (kind === "white") {
    35→    for (let i = 0; i < length; i++) data[i] = Math.random() * 2 - 1;
    36→  } else if (kind === "brown") {
    37→    let last = 0;
    38→    for (let i = 0; i < length; i++) {
    39→      const white = Math.random() * 2 - 1;
    40→      last = (last + 0.02 * white) / 1.02;
    41→      data[i] = last * 3.2;
    42→    }
    43→  } else {
    44→    // pink (Paul Kellet approximation)
    45→    let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
    46→    for (let i = 0; i < length; i++) {
    47→      const white = Math.random() * 2 - 1;
    48→      b0 = 0.99886 * b0 + white * 0.0555179;
    49→      b1 = 0.99332 * b1 + white * 0.0750759;
    50→      b2 = 0.969 * b2 + white * 0.153852;
    51→      b3 = 0.8665 * b3 + white * 0.3104856;
    52→      b4 = 0.55 * b4 + white * 0.5329522;
    53→      b5 = -0.7616 * b5 - white * 0.016898;
    54→      data[i] = (b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362) * 0.11;
    55→      b6 = white * 0.115926;
    56→    }
    57→  }
    58→  return buffer;
    59→}

    61→class Timers {
    62→  private timers: ReturnType<typeof setTimeout>[] = [];
    63→  after(ms: number, fn: () => void) {
    64→    const t = setTimeout(() => {
    65→      this.timers = this.timers.filter((x) => x !== t);
    66→      fn();
    67→    }, ms);
    68→    this.timers.push(t);
    69→    return t;
    70→  }
    71→  every(ms: number, fn: () => void) {
    72→    let alive = true;
    73→    const loop = () => {
    74→      if (!alive) return;
    75→      fn();
    76→      this.after(ms, loop);
    77→    };
    78→    this.after(ms, loop);
    79→    return () => {
    80→      alive = false;
    81→    };
    82→  }
    83→  dispose() {
    84→    this.timers.forEach(clearTimeout);
    85→    this.timers = [];
    86→  }
    87→}

    89→interface Sources {
    90→  nodes: AudioScheduledSourceNode[];

    92→function noiseSource(ctx: AudioContext, s: Sources, kind: "white" | "pink" | "brown"): AudioBufferSourceNode {
    93→  const src = ctx.createBufferSource();
    94→  src.buffer = makeNoiseBuffer(ctx, kind);
    95→  src.loop = true;
    96→  s.nodes.push(src);
    97→  return src;
    98→}

   100→/* one short filtered-noise burst (used for droplets, crackles, clinks…) */
   101→function burst(
   102→  ctx: AudioContext,
   103→  out: AudioNode,
   104→  opts: {
   105→    kind?: "white" | "pink" | "brown";
   106→    filter: { type: BiquadFilterType; freq: number; q?: number };
   107→    gain: number;
   108→    attack?: number;
   109→    decay: number;
   110→  }
   111→) {
   112→  const t = ctx.currentTime;
   113→  const src = ctx.createBufferSource();
   114→  src.buffer = makeNoiseBuffer(ctx, opts.kind ?? "white", 0.25);
   115→  const f = ctx.createBiquadFilter();
   116→  f.type = opts.filter.type;
   117→  f.frequency.value = opts.filter.freq;
   118→  if (opts.filter.q) f.Q.value = opts.filter.q;
   119→  const g = ctx.createGain();
   120→  g.gain.setValueAtTime(0.0001, t);
   121→  g.gain.exponentialRampToValueAtTime(Math.max(opts.gain, 0.0002), t + (opts.attack ?? 0.004));
   122→  g.gain.exponentialRampToValueAtTime(0.0001, t + opts.decay);
   123→  src.connect(f).connect(g).connect(out);
   124→  src.start(t, rnd(0, 2));
   125→  src.stop(t + opts.decay + 0.05);
   126→}

   128→/* one tonal ping with pitch envelope (birds, clinks, hoots, notes…) */
   129→function tone(
   130→  ctx: AudioContext,
   131→  out: AudioNode,
   132→  opts: {
   133→    type?: OscillatorType;
   134→    freq: number;
   135→    freqEnd?: number;
   136→    glide?: number;
   137→    gain: number;
   138→    attack?: number;
   139→    decay: number;
   140→    lowpass?: number;
   141→    detune?: number;
   142→  }
   143→) {
   144→  const t = ctx.currentTime;
   145→  const osc = ctx.createOscillator();
   146→  osc.type = opts.type ?? "sine";
   147→  osc.frequency.setValueAtTime(opts.freq, t);
   148→  if (opts.glide) {
   149→    osc.frequency.exponentialRampToValueAtTime(
   150→      Math.max(opts.freqEnd ?? opts.freq, 1),
   151→      t + opts.glide
   152→    );
   153→  }
   154→  if (opts.detune) osc.detune.value = opts.detune;
   155→  const g = ctx.createGain();
   156→  g.gain.setValueAtTime(0.0001, t);
   157→  g.gain.exponentialRampToValueAtTime(Math.max(opts.gain, 0.0002), t + (opts.attack ?? 0.01));
   158→  g.gain.exponentialRampToValueAtTime(0.0001, t + opts.decay);
   159→  let node: AudioNode = g;
   160→  if (opts.lowpass) {
   161→    const lp = ctx.createBiquadFilter();
   162→    lp.type = "lowpass";
   163→    lp.frequency.value = opts.lowpass;
   164→    g.connect(lp);
   165→    node = lp;
   166→  }
   167→  osc.connect(g);
   168→  node.connect(out);
   169→  osc.start(t);
   170→  osc.stop(t + opts.decay + 0.1);
   171→}

   175→const buildRain: Builder = (ctx, out) => {
   176→  const timers = new Timers();
   177→  const s: Sources = { nodes: [] };
   178→
   179→  // steady shower bed
   180→  const bed = noiseSource(ctx, s, "white");
   181→  const hp = ctx.createBiquadFilter();
   182→  hp.type = "highpass";
   183→  hp.frequency.value = 420;
   184→  const lp = ctx.createBiquadFilter();
   185→  lp.type = "lowpass";
   186→  lp.frequency.value = 6800;
   187→  const bedGain = ctx.createGain();
   188→  bedGain.gain.value = 0.3;
   189→  bed.connect(hp).connect(lp).connect(bedGain).connect(out);
   190→
   191→  // slow breathing of the shower intensity
   192→  const swell = ctx.createOscillator();
   193→  swell.frequency.value = 0.05;
   194→  const swellAmt = ctx.createGain();
   195→  swellAmt.gain.value = 0.06;
   196→  swell.connect(swellAmt).connect(bedGain.gain);
   197→  swell.start();
   198→  s.nodes.push(swell);
   199→
   200→  // individual droplet plips
   201→  const droplets = timers.every(rnd(90, 140), () => {
   202→    const n = Math.random() < 0.35 ? 2 : 1;
   203→    for (let i = 0; i < n; i++) {
   204→      timers.after(rnd(0, 120), () =>
   205→        burst(ctx, out, {
   206→          filter: { type: "bandpass", freq: rnd(700, 2600), q: rnd(4, 9) },
   207→          gain: rnd(0.015, 0.09),
   208→          decay: rnd(0.03, 0.09),
   209→        })
   210→      );
   211→    }
   212→  });
   213→
   214→  // far-off thunder, rare and soft
   215→  const thunder = timers.every(rnd(50000, 95000), () => {
   216→    const t = ctx.currentTime;
   217→    const src = ctx.createBufferSource();
   218→    src.buffer = makeNoiseBuffer(ctx, "brown", 6);
   219→    const lpT = ctx.createBiquadFilter();
   220→    lpT.type = "lowpass";
   221→    lpT.frequency.setValueAtTime(110, t);
   222→    lpT.frequency.exponentialRampToValueAtTime(45, t + 5);
   223→    const g = ctx.createGain();
   224→    g.gain.setValueAtTime(0.0001, t);
   225→    g.gain.exponentialRampToValueAtTime(0.11, t + 1.4);
   226→    g.gain.exponentialRampToValueAtTime(0.0001, t + 6);
   227→    src.connect(lpT).connect(g).connect(out);
   228→    src.start(t);
   229→    src.stop(t + 6.2);
   230→  });
   231→
   232→  return () => {
   233→    droplets();
   234→    thunder();
   235→    timers.dispose();
   236→    s.nodes.forEach((n) => {
   237→      try {
   238→        n.stop();
   239→      } catch { /* already stopped */ }
   240→    });
   241→  };
   242→};

   272→  // leaves catching the gusts
   273→  const leaves = noiseSource(ctx, s, "white");

   276→  bp.frequency.value = 5200;

   280→  leaves.connect(bp).connect(leafGain).connect(out);

   289→    g.gain.setTargetAtTime(peak, t, up / 3);
   290→    g.gain.setTargetAtTime(0.2, t + up, down / 3);

   292→    leafGain.gain.setTargetAtTime(0.014, t + up, down / 3);
   293→  });
   294→
   295→  s.nodes.push(lfo, lfo2);
   296→  return () => {
   297→    gusts();
   298→    timers.dispose();
   299→    s.nodes.forEach((n) => {
   300→      try {
   301→        n.stop();
   302→      } catch { /* noop */ }
   303→    });
   304→  };

   572→      );

   594→const PENTATONIC = [220, 261.63, 293.66, 329.63, 392, 440, 523.25, 587.33, 659.25];

   596→const buildPiano: Builder = (ctx, out) => {
   597→  const timers = new Timers();
   598→  const s: Sources = { nodes: [] };
   599→
   600→  // warm air bed
   601→  const air = noiseSource(ctx, s, "pink");
   602→  const lp = ctx.createBiquadFilter();
   603→  lp.type = "lowpass";
   604→  lp.frequency.value = 2600;
   605→  const airGain = ctx.createGain();
   606→  airGain.gain.value = 0.012;
   607→  air.connect(lp).connect(airGain).connect(out);
   608→
   609→  const padBus = ctx.createGain();
   610→  padBus.gain.value = 1;
   611→  const padLp = ctx.createBiquadFilter();
   612→  padLp.type = "lowpass";
   613→  padLp.frequency.value = 1600;
   614→  padBus.connect(padLp).connect(out);
   615→
   616→  const live: { gain: GainNode; nodes: AudioScheduledSourceNode[] }[] = [];
   617→
   618→  const playPadChord = () => {
   619→    // fade the previous chord out slowly
   620→    const previous = live.splice(0, live.length);
   621→    previous.forEach((ch) => {
   622→      const t = ctx.currentTime;
   623→      ch.gain.gain.setTargetAtTime(0.0001, t, 4.5);
   624→      timers.after(15000, () => {
   625→        ch.nodes.forEach((n) => {
   626→          try {
   627→            n.stop();
   628→          } catch { /* noop */ }
   629→        });
   630→        try {
   631→          ch.gain.disconnect();
   632→        } catch { /* noop */ }
   633→      });
   634→    });
   635→
   636→    const rootI = Math.floor(rnd(0, 4));
   637→    const chord = [PENTATONIC[rootI], PENTATONIC[rootI + 2], PENTATONIC[rootI + 4]];
   638→    const chordGain = ctx.createGain();
   639→    chordGain.gain.setValueAtTime(0.0001, ctx.currentTime);
   640→    chordGain.gain.setTargetAtTime(0.16, ctx.currentTime, 3.4);
   641→    chordGain.connect(padBus);
   642→
   643→    const chordNodes: AudioScheduledSourceNode[] = [];
   644→    chord.forEach((f) => {
   645→      [-4, 3].forEach((cents) => {
   646→        const osc = ctx.createOscillator();
   647→        osc.type = "sine";
   648→        osc.frequency.value = f;
   649→        osc.detune.value = cents;
   650→        const g = ctx.createGain();
   651→        g.gain.value = 0.5 / chord.length;
   652→        osc.connect(g).connect(chordGain);
   653→        osc.start();
   654→        chordNodes.push(osc);
   655→      });
   656→    });
   657→    live.push({ gain: chordGain, nodes: chordNodes });
   658→  };
   659→
   660→  playPadChord();
   661→  const padCycle = timers.every(rnd(13000, 18000), playPadChord);
   662→
   663→  // sparse felt-piano melody notes
   664→  const melody = timers.every(rnd(5200, 10000), () => {
   665→    const f = pick(PENTATONIC.filter((x) => x >= 329));
   666→    tone(ctx, out, {
   667→      type: "sine",
   668→      freq: f,
   669→      gain: rnd(0.03, 0.06),
   670→      attack: 0.02,
   671→      decay: rnd(2.8, 4.6),
   672→      lowpass: 1400,
   673→    });
   674→    tone(ctx, out, {
   675→      type: "sine",
   676→      freq: f * 2,
   677→      gain: rnd(0.006, 0.014),
   678→      attack: 0.02,
   679→      decay: rnd(1.6, 2.4),
   680→      lowpass: 2400,
   681→    });
   682→  });
   683→
   684→  return () => {
   685→    padCycle();
   686→    melody();
   687→    timers.dispose();
   688→    live.forEach((ch) =>
   689→      ch.nodes.forEach((n) => {
   690→        try {
   691→          n.stop();
   692→        } catch { /* noop */ }
   693→      })
   694→    );
   695→    s.nodes.forEach((n) => {
   696→      try {
   697→        n.stop();
   698→      } catch { /* noop */ }
   699→    });
   700→  };
   701→};

   703→const buildTrain: Builder = (ctx, out) => {

   719→  const rockAmt = ctx.createGain();
   720→  rockAmt.gain.value = 0.05;

   765→      osc.type = "sine";
   766→      osc.frequency.value = f;
   767→      osc.detune.value = rnd(-6, 6);
   768→      const g = ctx.createGain();
   769→      g.gain.setValueAtTime(0.0001, t);
   770→      g.gain.exponentialRampToValueAtTime(0.026, t + 0.9);
   771→      g.gain.setValueAtTime(0.026, t + 2.1);
   772→      g.gain.exponentialRampToValueAtTime(0.0001, t + 3.6);

   774→      lp.type = "lowpass";
   775→      lp.frequency.value = 520;
   776→      osc.connect(g).connect(lp).connect(out);
   777→      osc.start(t);
   778→      osc.stop(t + 3.8);
   779→    });
   780→  });
   781→
   782→  return () => {
   783→    horn();
   784→    timers.dispose();
   785→    s.nodes.forEach((n) => {
   786→      try {
   787→        n.stop();
   788→      } catch { /* noop */ }
   789→    });
   790→  };

   793→/* one struck singing bowl: inharmonic partials with long decay + gentle beating */
   794→function strikeBowl(ctx: AudioContext, out: AudioNode, baseFreq: number, velocity = 1) {
   795→  const t = ctx.currentTime;
   796→  const pan = ctx.createStereoPanner?.();
   797→  const destination = pan ? (pan.connect(out), pan) : out;
   798→  if (pan) pan.pan.value = rnd(-0.55, 0.55);
   799→
   800→  // classic bowl mode ratios (inharmonic, slightly stretched)
   801→  const modes: [number, number, number][] = [
   802→    // [ratio, gain, decaySeconds]
   803→    [1, 1, rnd(9, 13)],
   804→    [2.71, 0.42, rnd(7, 10)],
   805→    [5.18, 0.2, rnd(5, 8)],
   806→    [8.9, 0.09, rnd(3.5, 5.5)],
   807→  ];
   808→
   809→  modes.forEach(([ratio, gain, decay]) => {
   810→    // two slightly detuned oscillators per mode → slow audible beating
   811→    const beat = ratio === 1 ? rnd(0.6, 1.4) : rnd(1, 2.2);
   812→    for (const detune of [-beat, beat]) {
   813→      const osc = ctx.createOscillator();
   814→      osc.type = "sine";
   815→      osc.frequency.value = baseFreq * ratio;
   816→      osc.detune.value = detune * 100 * 0.06; // cents — a whisper off pitch
   817→      const g = ctx.createGain();
   818→      const amp = gain * velocity * 0.05;
   819→      g.gain.setValueAtTime(0.0001, t);
   820→      g.gain.exponentialRampToValueAtTime(Math.max(amp, 0.0002), t + rnd(0.03, 0.12));
   821→      g.gain.exponentialRampToValueAtTime(0.0001, t + decay);
   822→      osc.connect(g).connect(destination);
   823→      osc.start(t);
   824→      osc.stop(t + decay + 0.1);
   825→    }
   826→  });
   827→
   828→  // the mallet touch itself — a soft filtered knock
   829→  burst(ctx, destination, {
   830→    filter: { type: "bandpass", freq: baseFreq * 2.4, q: 1.4 },
   831→    gain: 0.05 * velocity,
   832→    decay: 0.14,
   833→  });
   834→}
   835→
   836→const buildBowls: Builder = (ctx, out) => {
   837→  const timers = new Timers();
   838→  const s: Sources = { nodes: [] };
   839→
   840→  // warm dark drone beneath everything (root + fifth, slow beating)
   841→  [110, 164.8].forEach((f, i) => {
   842→    for (const detune of [-4, 4]) {
   843→      const osc = ctx.createOscillator();
   844→      osc.type = "sine";
   845→      osc.frequency.value = f;
   846→      osc.detune.value = detune;
   847→      const g = ctx.createGain();
   848→      g.gain.value = i === 0 ? 0.028 : 0.016;
   849→      const lfo = ctx.createOscillator();
   850→      lfo.frequency.value = rnd(0.05, 0.09);
   851→      const lfoAmt = ctx.createGain();
   852→      lfoAmt.gain.value = g.gain.value * 0.45;
   853→      lfo.connect(lfoAmt).connect(g.gain);
   854→      osc.connect(g).connect(out);
   855→      osc.start();
   856→      lfo.start();
   857→      s.nodes.push(osc, lfo);
   858→    }
   859→  });
   860→
   861→  // faint stone-room shimmer: high-passed pink noise breathing
   862→  const air = noiseSource(ctx, s, "pink");
   863→  const hp = ctx.createBiquadFilter();
   864→  hp.type = "highpass";
   865→  hp.frequency.value = 2400;
   866→  const aGain = ctx.createGain();
   867→  aGain.gain.value = 0.012;
   868→  air.connect(hp).connect(aGain).connect(out);
   869→  const aLfo = ctx.createOscillator();
   870→  aLfo.frequency.value = 0.06;
   871→  const aLfoAmt = ctx.createGain();
   872→  aLfoAmt.gain.value = 0.007;
   873→  aLfo.connect(aLfoAmt).connect(aGain.gain);
   874→  aLfo.start();
   875→  s.nodes.push(aLfo);
   876→
   877→  // the bowls themselves — one every 9–16 s, occasionally a soft double-tap
   878→  const bowlNotes = [196, 220, 261.6, 293.7, 349.2, 392];
   879→  let lastNote = -1;
   880→  const scheduleBowl = () => {
   881→    let idx = Math.floor(Math.random() * bowlNotes.length);
   882→    if (idx === lastNote) idx = (idx + 1) % bowlNotes.length;
   883→    lastNote = idx;
   884→    strikeBowl(ctx, out, bowlNotes[idx], rnd(0.75, 1.1));
   885→    if (Math.random() < 0.22) {
   886→      timers.after(rnd(700, 1300), () => strikeBowl(ctx, out, bowlNotes[idx] * 1.5, 0.4));
   887→    }
   888→    timers.after(rnd(9000, 16000), scheduleBowl);
   889→  };
   890→  timers.after(rnd(400, 1200), scheduleBowl);
   891→
   892→  return () => {
   893→    timers.dispose();
   894→    s.nodes.forEach((n) => {
   895→      try {
   896→        n.stop();
   897→      } catch { /* noop */ }
   898→    });
   899→  };
   900→};

   902→/**
   903→ * Snowfall — the world wrapped in a blanket.
   904→ * A deep muffled hush (everything high is absorbed), slow wind leaning on
   905→ * the house, the occasional soft whump of snow letting go of a roof, and
   906→ * — very rarely — a tiny icy tick, like a crystal settling on the pane.
   907→ */
   908→const buildSnow: Builder = (ctx, out) => {
   909→  const timers = new Timers();
   910→  const s: Sources = { nodes: [] };
   911→
   912→  // the hush itself: brown noise behind a thick felt blanket
   913→  const hush = noiseSource(ctx, s, "brown");
   914→  const hushLp = ctx.createBiquadFilter();
   915→  hushLp.type = "lowpass";
   916→  hushLp.frequency.value = 380;
   917→  const hushGain = ctx.createGain();
   918→  hushGain.gain.value = 0.14;
   919→  hush.connect(hushLp).connect(hushGain).connect(out);
   920→  // the blanket itself breathes, very slowly
   921→  const hushLfo = ctx.createOscillator();
   922→  hushLfo.frequency.value = rnd(0.045, 0.07);
   923→  const hushLfoAmt = ctx.createGain();
   924→  hushLfoAmt.gain.value = 0.035;
   925→  hushLfo.connect(hushLfoAmt).connect(hushGain.gain);
   926→  hushLfo.start();
   927→  s.nodes.push(hushLfo);
   928→
   929→  // wind leaning on the walls — band-passed pink noise with a wandering center
   930→  const wind = noiseSource(ctx, s, "pink");
   931→  const windBp = ctx.createBiquadFilter();
   932→  windBp.type = "bandpass";
   933→  windBp.frequency.value = 300;
   934→  windBp.Q.value = 0.7;
   935→  const windGain = ctx.createGain();
   936→  windGain.gain.value = 0.03;
   937→  const windPan = ctx.createStereoPanner();
   938→  wind.connect(windBp).connect(windGain).connect(windPan).connect(out);
   939→  const windWander = ctx.createOscillator();
   940→  windWander.frequency.value = 0.05;
   941→  const windWanderAmt = ctx.createGain();
   942→  windWanderAmt.gain.value = 140;
   943→  windWander.connect(windWanderAmt).connect(windBp.frequency);
   944→  windWander.start();
   945→  s.nodes.push(windWander);
   946→
   947→  // gusts: the wind swells, leans from one side, then gives up
   948→  const gust = () => {
   949→    const t = ctx.currentTime;
   950→    const peak = rnd(0.035, 0.085);
   951→    const rise = rnd(2.2, 4.5);
   952→    const fall = rnd(4, 8);
   953→    windGain.gain.cancelScheduledValues(t);
   954→    windGain.gain.setTargetAtTime(peak, t, rise / 3);
   955→    windGain.gain.setTargetAtTime(0.012, t + rise, fall / 3);
   956→    windPan.pan.setTargetAtTime(rnd(-0.6, 0.6), t, rise / 2);
   957→    timers.after(rnd(14000, 26000), gust);
   958→  };
   959→  timers.after(rnd(2500, 6000), gust);
   960→
   961→  // snow letting go of the roof — a soft, rounded whump
   962→  const whump = () => {
   963→    burst(ctx, out, {
   964→      kind: "brown",
   965→      filter: { type: "lowpass", freq: rnd(180, 320) },
   966→      gain: rnd(0.10, 0.16),
   967→      attack: 0.22,
   968→      decay: rnd(0.9, 1.6),
   969→    });
   970→    // sometimes a second, smaller sigh from the far eave
   971→    if (Math.random() < 0.35) {
   972→      timers.after(rnd(700, 1600), () =>
   973→        burst(ctx, out, {
   974→          kind: "brown",
   975→          filter: { type: "lowpass", freq: rnd(140, 260) },
   976→          gain: rnd(0.05, 0.08),
   977→          attack: 0.3,
   978→          decay: rnd(0.8, 1.3),
   979→        })
   980→      );
   981→    }
   982→    timers.after(rnd(20000, 46000), whump);
   983→  };
   984→  timers.after(rnd(9000, 18000), whump);
   985→
   986→  // a rare icy tick — a crystal settling on the cold pane
   987→  const tickle = () => {
   988→    const pan = ctx.createStereoPanner();
   989→    pan.pan.value = rnd(-0.8, 0.8);
   990→    pan.connect(out);
   991→    tone(ctx, pan, {
   992→      type: "sine",
   993→      freq: rnd(1900, 3400),
   994→      gain: rnd(0.012, 0.022),
   995→      decay: rnd(0.25, 0.6),
   996→    });
   997→    timers.after(1200, () => {
   998→      try {
   999→        pan.disconnect();
  1000→      } catch { /* noop */ }
  1001→    });
  1002→    timers.after(rnd(17000, 40000), tickle);
  1003→  };
  1004→  timers.after(rnd(12000, 26000), tickle);
  1005→
  1006→  return () => {
  1007→    timers.dispose();
  1008→    s.nodes.forEach((n) => {
  1009→      try {
  1010→        n.stop();
  1011→      } catch { /* noop */ }
  1012→    });
  1013→  };
  1014→};

  1054→  private masterVolume = 0.9;
  1055→  private baseTrim = 1; // trim for the currently-loading base
  1056→  private baseTrims: Partial<Record<SoundscapeId, number>> = {}; // per-soundscape room level 0.4..1.2
  1057→
  1058→  get context(): AudioContext | null {
  1059→    return this.ctx;
  1060→  }
  1061→
  1062→  get currentBase(): SoundscapeId | null {
  1063→    return this.baseId;
  1064→  }

  1066→  ensureContext(): AudioContext {
  1067→    if (!this.ctx) {
  1068→      const AC: typeof AudioContext =
  1069→        window.AudioContext ||
  1070→        (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
  1071→      this.ctx = new AC();
  1072→      this.fade = this.ctx.createGain();
  1073→      this.fade.gain.value = 1;
  1074→      this.duck = this.ctx.createGain();
  1075→      this.duck.gain.value = 1;
  1076→      this.master = this.ctx.createGain();
  1077→      this.master.gain.value = this.masterVolume;
  1078→      this.analyser = this.ctx.createAnalyser();
  1079→      this.analyser.fftSize = 256;
  1080→      this.analyser.smoothingTimeConstant = 0.82;
  1081→      this.analyserData = new Uint8Array(new ArrayBuffer(this.analyser.frequencyBinCount));
  1082→      const comp = this.ctx.createDynamicsCompressor();
  1083→      comp.threshold.value = -20;
  1084→      comp.knee.value = 18;
  1085→      comp.ratio.value = 4;
  1086→      this.fade.connect(this.duck).connect(this.master).connect(comp).connect(this.ctx.destination);
  1087→      this.master.connect(this.analyser);
  1088→    }
  1089→    if (this.ctx.state === "suspended") void this.ctx.resume();
  1090→    return this.ctx;
  1091→  }

  1093→  /** start (or crossfade to) a featured soundscape */
  1094→  async playBase(id: SoundscapeId, fadeInSeconds = 2.4) {
  1095→    const ctx = this.ensureContext();
  1096→    if (this.baseId === id && this.base) return;
  1097→    if (this.base) {
  1098→      const old = this.base;
  1099→      old.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.9);
  1100→      setTimeout(() => {
  1101→        old.dispose();
  1102→        try {
  1103→          old.gain.disconnect();
  1104→        } catch { /* noop */ }
  1105→      }, 4200);
  1106→    }
  1107→    const gain = ctx.createGain();
  1108→    gain.gain.setValueAtTime(0.0001, ctx.currentTime);
  1109→    this.baseTrim = this.baseTrims[id] ?? 1;
  1110→    gain.gain.setTargetAtTime(this.baseTrim, ctx.currentTime, fadeInSeconds / 3);
  1111→    gain.connect(this.fade);
  1112→    const dispose = BASE_BUILDERS[id](ctx, gain);
  1113→    this.base = { gain, dispose };
  1114→    this.baseId = id;
  1115→  }
  1116→
  1117→  /** per-soundscape room level (0.4..1.2) — live-ramps only the matching active base */
  1118→  setBaseTrim(id: SoundscapeId, v: number) {
  1119→    const clamped = Math.max(0.4, Math.min(1.2, v));
  1120→    this.baseTrims[id] = clamped;
  1121→    if (this.ctx && this.base && this.baseId === id) {
  1122→      this.base.gain.gain.setTargetAtTime(clamped, this.ctx.currentTime, 0.25);
  1123→    }
  1124→  }
  1125→
  1126→  getTrim(id: SoundscapeId): number {
  1127→    return this.baseTrims[id] ?? 1;
  1128→  }
  1129→
  1130→  /** set a mixer layer level 0..1; builds/disposes lazily */
  1131→  setMixLayer(id: MixLayerId, level: number) {
  1132→    const ctx = this.ensureContext();
  1133→    const existing = this.mix[id];
  1134→    if (level <= 0.005) {
  1135→      if (existing) {
  1136→        existing.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.5);
  1137→        setTimeout(() => {
  1138→          existing.dispose();
  1139→          try {
  1140→            existing.gain.disconnect();
  1141→          } catch { /* noop */ }
  1142→        }, 2600);
  1143→        delete this.mix[id];
  1144→      }
  1145→      return;
  1146→    }
  1147→    if (!existing) {
  1148→      const gain = ctx.createGain();
  1149→      gain.gain.setValueAtTime(0.0001, ctx.currentTime);
  1150→      gain.gain.setTargetAtTime(level * MIX_SCALE[id], ctx.currentTime, 1.1);
  1151→      gain.connect(this.fade);
  1152→      const dispose = MIX_BUILDERS[id](ctx, gain);
  1153→      this.mix[id] = { gain, dispose };
  1154→    } else {
  1155→      existing.gain.gain.setTargetAtTime(level * MIX_SCALE[id], ctx.currentTime, 0.25);
  1156→    }
  1157→  }
  1158→
  1159→  setMasterVolume(v: number) {
  1160→    this.masterVolume = v;
  1161→    if (this.ctx) this.master.gain.setTargetAtTime(v, this.ctx.currentTime, 0.12);
  1162→  }

  1164→  /** timer fade — call each tick with remaining/duration when inside last minute */
  1165→  setFadeFactor(f: number, rampSeconds = 0.6) {
  1166→    if (!this.ctx) return;
  1167→    const clamped = Math.max(0.0001, Math.min(1, f));
  1168→    this.fade.gain.setTargetAtTime(clamped, this.ctx.currentTime, rampSeconds / 3);
  1169→  }
  1170→
  1171→  /** duck the bed under a narrated voice — 0 = full, 1 = eased to a quarter */
  1172→  setDuck(amount: number, rampSeconds = 1.4) {
  1173→    if (!this.ctx) return;
  1174→    const clamped = Math.max(0, Math.min(1, amount));
  1175→    this.duck.gain.setTargetAtTime(1 - clamped * 0.75, this.ctx.currentTime, rampSeconds / 3);
  1176→  }
  1177→
  1178→  resetFade() {
  1179→    if (!this.ctx) return;
  1180→    this.fade.gain.setTargetAtTime(1, this.ctx.currentTime, 0.8);
  1181→  }
  1182→
  1183→  stopAll(fadeSeconds = 1) {
  1184→    if (!this.ctx) return;
  1185→    this.fade.gain.setTargetAtTime(0.0001, this.ctx.currentTime, fadeSeconds / 3);
  1186→    const kill = () => {
  1187→      this.base?.dispose();
  1188→      this.base = null;
  1189→      this.baseId = null;
  1190→      Object.values(this.mix).forEach((l) => l?.dispose());
  1191→      this.mix = {};
  1192→      setTimeout(() => {
  1193→        try {
  1194→          this.fade.gain.setValueAtTime(1, this.ctx.currentTime);
  1195→        } catch { /* noop */ }
  1196→      }, 300);
  1197→    };
  1198→    setTimeout(kill, fadeSeconds * 1000 + 150);
  1199→  }

  1201→  get isSilent() {
  1202→    return !this.base;
  1203→  }

  1205→  /**
  1206→   * A soft bell, routed around the timer fade (straight to master).
  1207→   * kind "complete" — single low bell when a sleep timer runs its course.
  1208→   * kind "sunrise"  — gentle rising motif for the wake light, repeats 3×.
  1209→   */
  1210→  playChime(kind: "complete" | "sunrise" = "complete") {
  1211→    const ctx = this.ensureContext();
  1212→    if (ctx.state === "suspended") void ctx.resume();
  1213→    const bus = ctx.createGain();
  1214→    bus.gain.value = 0.5;
  1215→    bus.connect(this.master);
  1216→
  1217→    const bell = (freq: number, at: number, level: number, decay = 4.5) => {
  1218→      const partials = [
  1219→        { f: freq, g: 1 },
  1220→        { f: freq * 2.01, g: 0.34 },
  1221→        { f: freq * 2.99, g: 0.11 },
  1222→      ];
  1223→      partials.forEach(({ f, g }) => {
  1224→        const osc = ctx.createOscillator();
  1225→        osc.type = "sine";
  1226→        osc.frequency.value = f;
  1227→        const g1 = ctx.createGain();
  1228→        g1.gain.setValueAtTime(0.0001, at);
  1229→        g1.gain.exponentialRampToValueAtTime(level * g, at + 0.06);
  1230→        g1.gain.exponentialRampToValueAtTime(0.0001, at + decay);
  1231→        osc.connect(g1).connect(bus);
  1232→        osc.start(at);
  1233→        osc.stop(at + decay + 0.2);
  1234→      });
  1235→    };
  1236→
  1237→    const t0 = ctx.currentTime + 0.08;
  1238→    if (kind === "complete") {
  1239→      // A4 then E5 — "it is safe to sleep now"
  1240→      bell(432, t0, 0.2, 5);
  1241→      bell(648, t0 + 0.9, 0.13, 5.5);
  1242→    } else {
  1243→      // C5 · E5 · G5 rising, thrice, softer each time
  1244→      const motif = [523.25, 659.25, 783.99];
  1245→      for (let rep = 0; rep < 3; rep++) {
  1246→        const start = t0 + rep * 5.2;
  1247→        motif.forEach((f, i) => bell(f, start + i * 0.85, 0.16 * (1 - rep * 0.26), 4));
  1248→      }
  1249→    }
  1250→    // release the bus long after the last partial dies
  1251→    setTimeout(() => {
  1252→      try {
  1253→        bus.disconnect();
  1254→      } catch { /* noop */ }
  1255→    }, (kind === "complete" ? 8 : 22) * 1000);
  1256→  }

  1258→  /** RMS loudness 0..1 — for gentle audio-reactive visuals */
  1259→  getLevel(): number {
  1260→    if (!this.analyser || !this.analyserData || !this.ctx) return 0;
  1261→    this.analyser.getByteFrequencyData(this.analyserData);
  1262→    let sum = 0;
  1263→    for (let i = 0; i < this.analyserData.length; i++) sum += this.analyserData[i];
  1264→    return Math.min(1, sum / this.analyserData.length / 140);
  1265→  }
