    26→const rnd = (min: number, max: number) => min + Math.random() * (max - min);
    27→const pick = <T,>(arr: T[]): T => arr[Math.floor(Math.random() * arr.length)];

    29→function makeNoiseBuffer(ctx: AudioContext, kind: "white" | "pink" | "brown", seconds = 4): AudioBuffer {
    30→  const length = ctx.sampleRate * seconds;
    31→  const buffer = ctx.createBuffer(1, length, ctx.sampleRate);
    32→  const data = buffer.getChannelData(0);
    33→  if (kind === "white") {
    34→    for (let i = 0; i < length; i++) data[i] = Math.random() * 2 - 1;
    35→  } else if (kind === "brown") {
    36→    let last = 0;
    37→    for (let i = 0; i < length; i++) {
    38→      const white = Math.random() * 2 - 1;
    39→      last = (last + 0.02 * white) / 1.02;
    40→      data[i] = last * 3.2;
    41→    }
    42→  } else {
    43→    // pink (Paul Kellet approximation)
    44→    let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
    45→    for (let i = 0; i < length; i++) {
    46→      const white = Math.random() * 2 - 1;
    47→      b0 = 0.99886 * b0 + white * 0.0555179;
    48→      b1 = 0.99332 * b1 + white * 0.0750759;
    49→      b2 = 0.969 * b2 + white * 0.153852;
    50→      b3 = 0.8665 * b3 + white * 0.3104856;
    51→      b4 = 0.55 * b4 + white * 0.5329522;
    52→      b5 = -0.7616 * b5 - white * 0.016898;
    53→      data[i] = (b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362) * 0.11;
    54→      b6 = white * 0.115926;
    55→    }
    56→  }
    57→  return buffer;
    58→}

    60→class Timers {
    61→  private timers: ReturnType<typeof setTimeout>[] = [];
    62→  after(ms: number, fn: () => void) {
    63→    const t = setTimeout(() => {
    64→      this.timers = this.timers.filter((x) => x !== t);
    65→      fn();
    66→    }, ms);
    67→    this.timers.push(t);
    68→    return t;
    69→  }
    70→  every(ms: number, fn: () => void) {
    71→    let alive = true;
    72→    const loop = () => {
    73→      if (!alive) return;
    74→      fn();
    75→      this.after(ms, loop);
    76→    };
    77→    this.after(ms, loop);
    78→    return () => {
    79→      alive = false;
    80→    };
    81→  }
    82→  dispose() {
    83→    this.timers.forEach(clearTimeout);
    84→    this.timers = [];
    85→  }
    86→}

    88→interface Sources {
    89→  nodes: AudioScheduledSourceNode[];

    91→function noiseSource(ctx: AudioContext, s: Sources, kind: "white" | "pink" | "brown"): AudioBufferSourceNode {
    92→  const src = ctx.createBufferSource();
    93→  src.buffer = makeNoiseBuffer(ctx, kind);
    94→  src.loop = true;
    95→  s.nodes.push(src);
    96→  return src;
    97→}

    99→/* one short filtered-noise burst (used for droplets, crackles, clinks…) */
   100→function burst(
   101→  ctx: AudioContext,
   102→  out: AudioNode,
   103→  opts: {
   104→    kind?: "white" | "pink" | "brown";
   105→    filter: { type: BiquadFilterType; freq: number; q?: number };
   106→    gain: number;
   107→    attack?: number;
   108→    decay: number;
   109→  }
   110→) {
   111→  const t = ctx.currentTime;
   112→  const src = ctx.createBufferSource();
   113→  src.buffer = makeNoiseBuffer(ctx, opts.kind ?? "white", 0.25);
   114→  const f = ctx.createBiquadFilter();
   115→  f.type = opts.filter.type;
   116→  f.frequency.value = opts.filter.freq;
   117→  if (opts.filter.q) f.Q.value = opts.filter.q;
   118→  const g = ctx.createGain();
   119→  g.gain.setValueAtTime(0.0001, t);
   120→  g.gain.exponentialRampToValueAtTime(Math.max(opts.gain, 0.0002), t + (opts.attack ?? 0.004));
   121→  g.gain.exponentialRampToValueAtTime(0.0001, t + opts.decay);
   122→  src.connect(f).connect(g).connect(out);
   123→  src.start(t, rnd(0, 2));
   124→  src.stop(t + opts.decay + 0.05);
   125→}

   127→/* one tonal ping with pitch envelope (birds, clinks, hoots, notes…) */
   128→function tone(
   129→  ctx: AudioContext,
   130→  out: AudioNode,
   131→  opts: {
   132→    type?: OscillatorType;
   133→    freq: number;
   134→    freqEnd?: number;
   135→    glide?: number;
   136→    gain: number;
   137→    attack?: number;
   138→    decay: number;
   139→    lowpass?: number;
   140→    detune?: number;
   141→  }
   142→) {
   143→  const t = ctx.currentTime;
   144→  const osc = ctx.createOscillator();
   145→  osc.type = opts.type ?? "sine";
   146→  osc.frequency.setValueAtTime(opts.freq, t);
   147→  if (opts.glide) {
   148→    osc.frequency.exponentialRampToValueAtTime(
   149→      Math.max(opts.freqEnd ?? opts.freq, 1),
   150→      t + opts.glide
   151→    );
   152→  }
   153→  if (opts.detune) osc.detune.value = opts.detune;
   154→  const g = ctx.createGain();
   155→  g.gain.setValueAtTime(0.0001, t);
   156→  g.gain.exponentialRampToValueAtTime(Math.max(opts.gain, 0.0002), t + (opts.attack ?? 0.01));
   157→  g.gain.exponentialRampToValueAtTime(0.0001, t + opts.decay);
   158→  let node: AudioNode = g;
   159→  if (opts.lowpass) {
   160→    const lp = ctx.createBiquadFilter();
   161→    lp.type = "lowpass";
   162→    lp.frequency.value = opts.lowpass;
   163→    g.connect(lp);
   164→    node = lp;
   165→  }
   166→  osc.connect(g);
   167→  node.connect(out);
   168→  osc.start(t);
   169→  osc.stop(t + opts.decay + 0.1);
   170→}

   174→const buildRain: Builder = (ctx, out) => {
   175→  const timers = new Timers();
   176→  const s: Sources = { nodes: [] };
   177→
   178→  // steady shower bed
   179→  const bed = noiseSource(ctx, s, "white");
   180→  const hp = ctx.createBiquadFilter();
   181→  hp.type = "highpass";
   182→  hp.frequency.value = 420;
   183→  const lp = ctx.createBiquadFilter();
   184→  lp.type = "lowpass";
   185→  lp.frequency.value = 6800;
   186→  const bedGain = ctx.createGain();
   187→  bedGain.gain.value = 0.3;
   188→  bed.connect(hp).connect(lp).connect(bedGain).connect(out);
   189→
   190→  // slow breathing of the shower intensity
   191→  const swell = ctx.createOscillator();
   192→  swell.frequency.value = 0.05;
   193→  const swellAmt = ctx.createGain();
   194→  swellAmt.gain.value = 0.06;
   195→  swell.connect(swellAmt).connect(bedGain.gain);
   196→  swell.start();
   197→  s.nodes.push(swell);
   198→
   199→  // individual droplet plips
   200→  const droplets = timers.every(rnd(90, 140), () => {
   201→    const n = Math.random() < 0.35 ? 2 : 1;
   202→    for (let i = 0; i < n; i++) {
   203→      timers.after(rnd(0, 120), () =>
   204→        burst(ctx, out, {
   205→          filter: { type: "bandpass", freq: rnd(700, 2600), q: rnd(4, 9) },
   206→          gain: rnd(0.015, 0.09),
   207→          decay: rnd(0.03, 0.09),
   208→        })
   209→      );
   210→    }
   211→  });
   212→
   213→  // far-off thunder, rare and soft
   214→  const thunder = timers.every(rnd(50000, 95000), () => {
   215→    const t = ctx.currentTime;
   216→    const src = ctx.createBufferSource();
   217→    src.buffer = makeNoiseBuffer(ctx, "brown", 6);
   218→    const lpT = ctx.createBiquadFilter();
   219→    lpT.type = "lowpass";
   220→    lpT.frequency.setValueAtTime(110, t);
   221→    lpT.frequency.exponentialRampToValueAtTime(45, t + 5);
   222→    const g = ctx.createGain();
   223→    g.gain.setValueAtTime(0.0001, t);
   224→    g.gain.exponentialRampToValueAtTime(0.11, t + 1.4);
   225→    g.gain.exponentialRampToValueAtTime(0.0001, t + 6);
   226→    src.connect(lpT).connect(g).connect(out);
   227→    src.start(t);
   228→    src.stop(t + 6.2);
   229→  });
   230→
   231→  return () => {
   232→    droplets();
   233→    thunder();
   234→    timers.dispose();
   235→    s.nodes.forEach((n) => {
   236→      try {
   237→        n.stop();
   238→      } catch { /* already stopped */ }
   239→    });
   240→  };
   241→};

   294→  s.nodes.push(lfo, lfo2);

   296→    gusts();
   297→    timers.dispose();
   298→    s.nodes.forEach((n) => {
   299→      try {
   300→        n.stop();
   301→      } catch { /* noop */ }
   302→    });
   303→  };

   792→/* one struck singing bowl: inharmonic partials with long decay + gentle beating */
   793→function strikeBowl(ctx: AudioContext, out: AudioNode, baseFreq: number, velocity = 1) {
   794→  const t = ctx.currentTime;
   795→  const pan = ctx.createStereoPanner?.();
   796→  const destination = pan ? (pan.connect(out), pan) : out;
   797→  if (pan) pan.pan.value = rnd(-0.55, 0.55);
   798→
   799→  // classic bowl mode ratios (inharmonic, slightly stretched)
   800→  const modes: [number, number, number][] = [
   801→    // [ratio, gain, decaySeconds]
   802→    [1, 1, rnd(9, 13)],
   803→    [2.71, 0.42, rnd(7, 10)],
   804→    [5.18, 0.2, rnd(5, 8)],
   805→    [8.9, 0.09, rnd(3.5, 5.5)],
   806→  ];
   807→
   808→  modes.forEach(([ratio, gain, decay]) => {
   809→    // two slightly detuned oscillators per mode → slow audible beating
   810→    const beat = ratio === 1 ? rnd(0.6, 1.4) : rnd(1, 2.2);
   811→    for (const detune of [-beat, beat]) {
   812→      const osc = ctx.createOscillator();
   813→      osc.type = "sine";
   814→      osc.frequency.value = baseFreq * ratio;
   815→      osc.detune.value = detune * 100 * 0.06; // cents — a whisper off pitch
   816→      const g = ctx.createGain();
   817→      const amp = gain * velocity * 0.05;
   818→      g.gain.setValueAtTime(0.0001, t);
   819→      g.gain.exponentialRampToValueAtTime(Math.max(amp, 0.0002), t + rnd(0.03, 0.12));
   820→      g.gain.exponentialRampToValueAtTime(0.0001, t + decay);
   821→      osc.connect(g).connect(destination);
   822→      osc.start(t);
   823→      osc.stop(t + decay + 0.1);
   824→    }
   825→  });
   826→
   827→  // the mallet touch itself — a soft filtered knock
   828→  burst(ctx, destination, {
   829→    filter: { type: "bandpass", freq: baseFreq * 2.4, q: 1.4 },
   830→    gain: 0.05 * velocity,
   831→    decay: 0.14,
   832→  });
   833→}
   834→
   835→const buildBowls: Builder = (ctx, out) => {
   836→  const timers = new Timers();
   837→  const s: Sources = { nodes: [] };
   838→
   839→  // warm dark drone beneath everything (root + fifth, slow beating)
   840→  [110, 164.8].forEach((f, i) => {
   841→    for (const detune of [-4, 4]) {
   842→      const osc = ctx.createOscillator();
   843→      osc.type = "sine";
   844→      osc.frequency.value = f;
   845→      osc.detune.value = detune;
   846→      const g = ctx.createGain();
   847→      g.gain.value = i === 0 ? 0.028 : 0.016;
   848→      const lfo = ctx.createOscillator();
   849→      lfo.frequency.value = rnd(0.05, 0.09);
   850→      const lfoAmt = ctx.createGain();
   851→      lfoAmt.gain.value = g.gain.value * 0.45;
   852→      lfo.connect(lfoAmt).connect(g.gain);
   853→      osc.connect(g).connect(out);
   854→      osc.start();
   855→      lfo.start();
   856→      s.nodes.push(osc, lfo);
   857→    }
   858→  });
   859→
   860→  // faint stone-room shimmer: high-passed pink noise breathing
   861→  const air = noiseSource(ctx, s, "pink");
   862→  const hp = ctx.createBiquadFilter();
   863→  hp.type = "highpass";
   864→  hp.frequency.value = 2400;
   865→  const aGain = ctx.createGain();
   866→  aGain.gain.value = 0.012;
   867→  air.connect(hp).connect(aGain).connect(out);
   868→  const aLfo = ctx.createOscillator();
   869→  aLfo.frequency.value = 0.06;
   870→  const aLfoAmt = ctx.createGain();
   871→  aLfoAmt.gain.value = 0.007;
   872→  aLfo.connect(aLfoAmt).connect(aGain.gain);
   873→  aLfo.start();
   874→  s.nodes.push(aLfo);

   876→  // the bowls themselves — one every 9–16 s, occasionally a soft double-tap
   877→  const bowlNotes = [196, 220, 261.6, 293.7, 349.2, 392];
   878→  let lastNote = -1;
   879→  const scheduleBowl = () => {
   880→    let idx = Math.floor(Math.random() * bowlNotes.length);
   881→    if (idx === lastNote) idx = (idx + 1) % bowlNotes.length;
   882→    lastNote = idx;
   883→    strikeBowl(ctx, out, bowlNotes[idx], rnd(0.75, 1.1));
   884→    if (Math.random() < 0.22) {
   885→      timers.after(rnd(700, 1300), () => strikeBowl(ctx, out, bowlNotes[idx] * 1.5, 0.4));
   886→    }
   887→    timers.after(rnd(9000, 16000), scheduleBowl);
   888→  };
   889→  timers.after(rnd(400, 1200), scheduleBowl);
   890→
   891→  return () => {
   892→    timers.dispose();
   893→    s.nodes.forEach((n) => {
   894→      try {
   895→        n.stop();
   896→      } catch { /* noop */ }
   897→    });
   898→  };
   899→};

   912→const MIX_BUILDERS: Record<MixLayerId, Builder> = {
   913→  rain: buildRain,

   918→/* mix layers are a touch quieter than the featured soundscape */
   919→const MIX_SCALE: Record<MixLayerId, number> = { rain: 0.85, wind: 0.9, fire: 0.8 };

   921→/* ─────────────────────────── the engine ──────────────────────── */

   923→interface LayerState {
   924→  gain: GainNode;
   925→  dispose: Dispose;
   926→}

   928→export class AudioEngine {
   929→  private ctx: AudioContext | null = null;
   930→  private fade!: GainNode;
   931→  private duck!: GainNode;
   932→  private master!: GainNode;
   933→  private analyser: AnalyserNode | null = null;
   934→  private analyserData: Uint8Array<ArrayBuffer> | null = null;
   935→  private base: LayerState | null = null;
   936→  private baseId: SoundscapeId | null = null;
   937→  private mix: Partial<Record<MixLayerId, LayerState>> = {};
   938→  private masterVolume = 0.9;
   939→  private baseTrim = 1; // trim for the currently-loading base
   940→  private baseTrims: Partial<Record<SoundscapeId, number>> = {}; // per-soundscape room level 0.4..1.2
   941→
   942→  get context(): AudioContext | null {
   943→    return this.ctx;
   944→  }
   945→
   946→  get currentBase(): SoundscapeId | null {
   947→    return this.baseId;
   948→  }
   949→
   950→  ensureContext(): AudioContext {
   951→    if (!this.ctx) {
   952→      const AC: typeof AudioContext =
   953→        window.AudioContext ||
   954→        (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   955→      this.ctx = new AC();
   956→      this.fade = this.ctx.createGain();
   957→      this.fade.gain.value = 1;
   958→      this.duck = this.ctx.createGain();
   959→      this.duck.gain.value = 1;
   960→      this.master = this.ctx.createGain();
   961→      this.master.gain.value = this.masterVolume;
   962→      this.analyser = this.ctx.createAnalyser();
   963→      this.analyser.fftSize = 256;
   964→      this.analyser.smoothingTimeConstant = 0.82;
   965→      this.analyserData = new Uint8Array(new ArrayBuffer(this.analyser.frequencyBinCount));
   966→      const comp = this.ctx.createDynamicsCompressor();
   967→      comp.threshold.value = -20;
   968→      comp.knee.value = 18;
   969→      comp.ratio.value = 4;
   970→      this.fade.connect(this.duck).connect(this.master).connect(comp).connect(this.ctx.destination);
   971→      this.master.connect(this.analyser);
   972→    }
   973→    if (this.ctx.state === "suspended") void this.ctx.resume();
   974→    return this.ctx;
   975→  }
   976→
   977→  /** start (or crossfade to) a featured soundscape */
   978→  async playBase(id: SoundscapeId, fadeInSeconds = 2.4) {
   979→    const ctx = this.ensureContext();
   980→    if (this.baseId === id && this.base) return;
   981→    if (this.base) {
   982→      const old = this.base;
   983→      old.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.9);
   984→      setTimeout(() => {
   985→        old.dispose();
   986→        try {
   987→          old.gain.disconnect();
   988→        } catch { /* noop */ }
   989→      }, 4200);
   990→    }
   991→    const gain = ctx.createGain();
   992→    gain.gain.setValueAtTime(0.0001, ctx.currentTime);
   993→    this.baseTrim = this.baseTrims[id] ?? 1;
   994→    gain.gain.setTargetAtTime(this.baseTrim, ctx.currentTime, fadeInSeconds / 3);
   995→    gain.connect(this.fade);
   996→    const dispose = BASE_BUILDERS[id](ctx, gain);
   997→    this.base = { gain, dispose };
   998→    this.baseId = id;
   999→  }
  1000→
  1001→  /** per-soundscape room level (0.4..1.2) — live-ramps only the matching active base */
  1002→  setBaseTrim(id: SoundscapeId, v: number) {
  1003→    const clamped = Math.max(0.4, Math.min(1.2, v));
  1004→    this.baseTrims[id] = clamped;
  1005→    if (this.ctx && this.base && this.baseId === id) {
  1006→      this.base.gain.gain.setTargetAtTime(clamped, this.ctx.currentTime, 0.25);
  1007→    }
  1008→  }
  1009→
  1010→  getTrim(id: SoundscapeId): number {
  1011→    return this.baseTrims[id] ?? 1;
  1012→  }
  1013→
  1014→  /** set a mixer layer level 0..1; builds/disposes lazily */
  1015→  setMixLayer(id: MixLayerId, level: number) {
  1016→    const ctx = this.ensureContext();
  1017→    const existing = this.mix[id];
  1018→    if (level <= 0.005) {
  1019→      if (existing) {
  1020→        existing.gain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.5);
  1021→        setTimeout(() => {
  1022→          existing.dispose();
  1023→          try {
  1024→            existing.gain.disconnect();
  1025→          } catch { /* noop */ }
  1026→        }, 2600);
  1027→        delete this.mix[id];
  1028→      }
  1029→      return;
  1030→    }
  1031→    if (!existing) {
  1032→      const gain = ctx.createGain();
  1033→      gain.gain.setValueAtTime(0.0001, ctx.currentTime);
  1034→      gain.gain.setTargetAtTime(level * MIX_SCALE[id], ctx.currentTime, 1.1);
  1035→      gain.connect(this.fade);
  1036→      const dispose = MIX_BUILDERS[id](ctx, gain);
  1037→      this.mix[id] = { gain, dispose };
  1038→    } else {
  1039→      existing.gain.gain.setTargetAtTime(level * MIX_SCALE[id], ctx.currentTime, 0.25);
  1040→    }
  1041→  }
  1042→
  1043→  setMasterVolume(v: number) {
  1044→    this.masterVolume = v;
  1045→    if (this.ctx) this.master.gain.setTargetAtTime(v, this.ctx.currentTime, 0.12);
  1046→  }
  1047→
  1048→  /** timer fade — call each tick with remaining/duration when inside last minute */
  1049→  setFadeFactor(f: number, rampSeconds = 0.6) {
  1050→    if (!this.ctx) return;
  1051→    const clamped = Math.max(0.0001, Math.min(1, f));
  1052→    this.fade.gain.setTargetAtTime(clamped, this.ctx.currentTime, rampSeconds / 3);
  1053→  }

  1055→  /** duck the bed under a narrated voice — 0 = full, 1 = eased to a quarter */

  1089→  /**
  1090→   * A soft bell, routed around the timer fade (straight to master).
  1091→   * kind "complete" — single low bell when a sleep timer runs its course.
  1092→   * kind "sunrise"  — gentle rising motif for the wake light, repeats 3×.
  1093→   */
  1094→  playChime(kind: "complete" | "sunrise" = "complete") {
  1095→    const ctx = this.ensureContext();
  1096→    if (ctx.state === "suspended") void ctx.resume();
  1097→    const bus = ctx.createGain();
  1098→    bus.gain.value = 0.5;
  1099→    bus.connect(this.master);
  1100→
  1101→    const bell = (freq: number, at: number, level: number, decay = 4.5) => {
  1102→      const partials = [
  1103→        { f: freq, g: 1 },
  1104→        { f: freq * 2.01, g: 0.34 },
  1105→        { f: freq * 2.99, g: 0.11 },
  1106→      ];
  1107→      partials.forEach(({ f, g }) => {
  1108→        const osc = ctx.createOscillator();
  1109→        osc.type = "sine";
  1110→        osc.frequency.value = f;
  1111→        const g1 = ctx.createGain();
  1112→        g1.gain.setValueAtTime(0.0001, at);
  1113→        g1.gain.exponentialRampToValueAtTime(level * g, at + 0.06);
  1114→        g1.gain.exponentialRampToValueAtTime(0.0001, at + decay);
  1115→        osc.connect(g1).connect(bus);
  1116→        osc.start(at);
  1117→        osc.stop(at + decay + 0.2);
  1118→      });
  1119→    };
  1120→
  1121→    const t0 = ctx.currentTime + 0.08;
  1122→    if (kind === "complete") {
  1123→      // A4 then E5 — "it is safe to sleep now"
  1124→      bell(432, t0, 0.2, 5);
  1125→      bell(648, t0 + 0.9, 0.13, 5.5);
  1126→    } else {
  1127→      // C5 · E5 · G5 rising, thrice, softer each time
  1128→      const motif = [523.25, 659.25, 783.99];
  1129→      for (let rep = 0; rep < 3; rep++) {
  1130→        const start = t0 + rep * 5.2;
  1131→        motif.forEach((f, i) => bell(f, start + i * 0.85, 0.16 * (1 - rep * 0.26), 4));
  1132→      }
  1133→    }
  1134→    // release the bus long after the last partial dies
  1135→    setTimeout(() => {
  1136→      try {
  1137→        bus.disconnect();
  1138→      } catch { /* noop */ }
  1139→    }, (kind === "complete" ? 8 : 22) * 1000);
  1140→  }

  1142→  /** RMS loudness 0..1 — for gentle audio-reactive visuals */
  1143→  getLevel(): number {
  1144→    if (!this.analyser || !this.analyserData || !this.ctx) return 0;
  1145→    this.analyser.getByteFrequencyData(this.analyserData);
  1146→    let sum = 0;
  1147→    for (let i = 0; i < this.analyserData.length; i++) sum += this.analyserData[i];
  1148→    return Math.min(1, sum / this.analyserData.length / 140);
  1149→  }
