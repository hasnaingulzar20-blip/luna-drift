    50→function formatRemaining(sec: number) {
    51→  const m = Math.floor(sec / 60);
    52→  const s = sec % 60;
    53→  return `${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    54→}

    56→const SILENCE_LABEL = "Silence";

    58→function stepLabel(sound: SequenceSound): string {

    60→}

    76→      ? "bg-moon-200/15 text-moon-100 ring-moon-200/50 shadow-[0_0_18px_rgba(236,226,200,0.18)]"

    84→      {step.soundscape === "silence" ? (
    85→        <Moon className="h-3 w-3 text-mist-500" aria-hidden="true" />
