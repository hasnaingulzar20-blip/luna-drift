    53→function relTime(iso: string) {
    54→  const diff = Date.now() - new Date(iso).getTime();
    55→  const mins = Math.round(diff / 60000);
    56→  if (mins < 1) return "just now";
    57→  if (mins < 60) return `${mins}m ago`;
    58→  const hours = Math.round(mins / 60);
    59→  if (hours < 24) return `${hours}h ago`;
    60→  const days = Math.round(hours / 24);
    61→  return days === 1 ? "yesterday" : `${days}d ago`;
    62→}

    72→/** download the whole ledger as CSV */
    73→async function exportLedger() {
    74→  try {
    75→    const res = await fetch("/api/sessions", { cache: "no-store" });
    76→    if (!res.ok) return;
    77→    const json = (await res.json()) as {
    78→      sessions: { soundscape: string; minutes: number; completed: boolean; endedAt: string }[];
    79→    };
    80→    const esc = (v: string | number | boolean) => {
    81→      const s = String(v);
    82→      return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
    83→    };
    84→    const lines = [
    85→      "date,soundscape,minutes,completed",
    86→      ...json.sessions.map((s) =>
    87→        [esc(new Date(s.endedAt).toISOString().slice(0, 16).replace("T", " ")), esc(s.soundscape), s.minutes, s.completed].join(",")
    88→      ),
    89→    ];
    90→    const blob = new Blob(["\uFEFF" + lines.join("\n")], { type: "text/csv;charset=utf-8" });
    91→    const url = URL.createObjectURL(blob);
    92→    const a = document.createElement("a");
    93→    a.href = url;
    94→    a.download = "luna-drift-ledger.csv";
    95→    a.click();
    96→    URL.revokeObjectURL(url);
    97→  } catch {
    98→    /* the ledger can wait */
    99→  }
   100→}

   102→/** build the gentle weekly digest text and copy it to the clipboard */
   103→function buildDigest(data: ProfileData): string {
   104→  const week = data.week ?? [];
   105→  const nights = week.filter((d) => d.minutes > 0).length;
   106→  const weekTotal = week.reduce((sum, d) => sum + d.minutes, 0);
   107→  const best = week.reduce<{ day: string; minutes: number } | null>(
   108→    (b, d) => (d.minutes > 0 && (!b || d.minutes > b.minutes) ? d : b),
   109→    null
   110→  );
   111→  const hours = Math.floor(data.totalMinutes / 60);
   112→  const rest = data.totalMinutes % 60;
   113→  const top = data.insights?.topSoundscape ? prettyName(data.insights.topSoundscape) : null;
   114→  const lines = [
   115→    "the week that was — Luna Drift",
   116→    `· ${nights} night${nights === 1 ? "" : "s"} with rest`,
   117→  ];
   118→  if (best) lines.push(`· best night ${best.day}, ${best.minutes} min`);
   119→  if (top) lines.push(`· most heard: ${top}`);
   120→  if (data.streak > 0) lines.push(`· a ${data.streak}-night streak, still burning`);
   121→  lines.push(`· ${hours}h ${rest}m altogether`);
   122→  if (weekTotal > 0) lines.push(`· ${weekTotal} min this week`);
   123→  return lines.join("\n");
   124→}
